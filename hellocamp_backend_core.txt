import { NextResponse } from 'next/server';
import Stripe from 'stripe';
import { createClient } from '@supabase/supabase-js';

// Inicializa o Supabase do lado do servidor para gravar o ID com segurança
const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function POST(req: Request) {
  const stripeSecretKey = process.env.STRIPE_SECRET_KEY;
  
  if (!stripeSecretKey) {
    return NextResponse.json({ error: "Chave Secreta da Stripe não configurada no servidor." }, { status: 500 });
  }

  const stripe = new Stripe(stripeSecretKey);

  try {
    const body = await req.json();
    const { userId, email, lang } = body;
    const siteUrl = process.env.NEXT_PUBLIC_SITE_URL || 'https://www.hellocamp.pt';

    // 1. Procurar se o perfil já tem uma conta Stripe Connect associada
    const { data: perfil } = await supabaseAdmin
      .from('perfis')
      .select('stripe_account_id')
      .eq('id', userId)
      .single();

    let stripeAccountId = perfil?.stripe_account_id;

    // 2. Se não tiver, cria uma conta Express nova na Stripe
    if (!stripeAccountId) {
      const account = await stripe.accounts.create({
        type: 'express',
        email: email,
        capabilities: {
          card_payments: { requested: true },
          transfers: { requested: true },
        },
      });
      stripeAccountId = account.id;

      // Grava o novo ID imediatamente na tabela perfis do utilizador
      await supabaseAdmin
        .from('perfis')
        .update({ stripe_account_id: stripeAccountId })
        .eq('id', userId);
    }

    // 3. Gerar o link oficial de Onboarding (onde o parceiro vai colocar os dados bancários com segurança)
    const accountLink = await stripe.accountLinks.create({
      account: stripeAccountId,
      refresh_url: `${siteUrl}/${lang}/dashboard/faturacao`, // Se falhar ou expirar, volta aqui
      return_url: `${siteUrl}/${lang}/dashboard/faturacao`,  // Se concluir com sucesso, volta aqui
      type: 'account_onboarding',
    });

    return NextResponse.json({ url: accountLink.url });
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

import { NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;
const supabaseServiceKey = process.env.SUPABASE_SERVICE_ROLE_KEY!; // Utilize a service role key no .env para garantir permissões de escrita

const supabase = createClient(supabaseUrl, supabaseServiceKey);

async function traduzir(texto: string): Promise<string> {
  if (!texto) return "";
  try {
    const res = await fetch(`https://api.mymemory.translated.net/get?q=${encodeURIComponent(texto)}&langpair=pt|en`);
    const data = await res.json();
    return data.responseData.translatedText || texto;
  } catch (e) {
    return texto;
  }
}

export async function POST(request: Request) {
  try {
    const { id } = await request.json();
    if (!id) return NextResponse.json({ error: "ID do campo em falta" }, { status: 400 });

    // 1. Procura os dados originais em Português
    const { data: campo, error: fetchError } = await supabase.from("campos").select("*").eq("id", id).single();
    if (fetchError || !campo) return NextResponse.json({ error: "Campo não encontrado" }, { status: 404 });

    // 2. Executa as traduções em segundo plano no servidor
    const [
      nome_en, categoria_en, local_en, idade_en, descricao_en,
      alimentacao_en, alojamento_en, seguro_en, Distrito_en, regras_termos_en, politica_cancelamento_en
    ] = await Promise.all([
      traduzir(campo.nome), traduzir(campo.categoria), traduzir(campo.local),
      traduzir(campo.idade), traduzir(campo.descricao), traduzir(campo.alimentacao),
      traduzir(campo.alojamento), traduzir(campo.seguro), traduzir(campo.Distrito),
      traduzir(campo.regras_termos), traduzir(campo.politica_cancelamento)
    ]);

    // Tradução dos turnos
    let turnos_en = [];
    if (campo.turnos && Array.isArray(campo.turnos)) {
      turnos_en = await Promise.all(campo.turnos.map(async (t: any) => ({
        ...t,
        nome: await traduzir(t.nome)
      })));
    }

    // Tradução das perguntas customizadas
    let perguntasCustomizadasEn = [];
    if (campo.perguntas_customizadas && Array.isArray(campo.perguntas_customizadas)) {
      perguntasCustomizadasEn = await Promise.all(campo.perguntas_customizadas.map(async (p: string) => await traduzir(p)));
    }

    const textoDatasEn = campo.datas_disponiveis ? campo.datas_disponiveis.replace(/ a /g, " to ") : "";

    // 3. Atualiza as colunas de inglês na base de dados
    await supabase.from("campos").update({
      nome_en, categoria_en, local_en, idade_en, descricao_en,
      alimentacao_en, alojamento_en, seguro_en, Distrito_en,
      regras_termos_en, politica_cancelamento_en, turnos_en,
      perguntas_customizadas_en: perguntasCustomizadasEn,
      datas_disponiveis_en: textoDatasEn,
      pais_en: campo.pais === "Portugal" ? "Portugal" : "International"
    }).eq("id", id);

    return NextResponse.json({ success: true });
  } catch (error: any) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

import { NextResponse } from 'next/server';
import { createClient } from '@supabase/supabase-js';

export const dynamic = 'force-dynamic';

const supabaseAdmin = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function POST(req: Request) {
  try {
    // 1. Opcional: Verificação de Segurança
    const tokenHeader = req.headers.get("auth-token");
    const secret = process.env.BREVO_WEBHOOK_SECRET;

    if (secret && tokenHeader !== secret) {
      return NextResponse.json({ error: "Acesso não autorizado." }, { status: 401 });
    }

    const payload = await req.json();

    // 2. Extrair dados da Brevo
    const eventType = payload.event; 
    // Na Brevo, tags vêm como array. A primeira (index 0) é o broadcastId que injetamos na rota de envio.
    const broadcastId = payload.tags && payload.tags.length > 0 ? payload.tags[0] : null;

    if (!broadcastId) {
      return NextResponse.json({ message: "Evento ignorado: Tag não encontrada." }, { status: 200 });
    }

    // 3. Mapear os eventos para as colunas
    let updateField = '';
    if (eventType === 'delivered') updateField = 'total_entregues';
    else if (eventType === 'opened' || eventType === 'unique_opened') updateField = 'total_abertos';
    else if (eventType === 'click') updateField = 'total_cliques';
    else if (['bounced', 'hardBounced', 'softBounced', 'spam', 'invalid', 'blocked', 'deferred'].includes(eventType)) {
      updateField = 'total_falhas';
    }

    // 4. Atualizar a base de dados
    if (updateField) {
      const { data: campanha, error: fetchError } = await supabaseAdmin
        .from('email_broadcasts')
        .select(updateField)
        .eq('id', broadcastId)
        .single();

      if (!fetchError && campanha) {
        const valorAtual = (campanha as Record<string, any>)?.[updateField] || 0;
        
        await supabaseAdmin
          .from('email_broadcasts')
          .update({ [updateField]: valorAtual + 1 })
          .eq('id', broadcastId);
      }
    }

    return NextResponse.json({ received: true }, { status: 200 });

  } catch (err: any) {
    console.error("Erro no Webhook da Brevo:", err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function POST(req: Request) {
  try {
    // 1. VERIFICAÇÃO DE SEGURANÇA (Estilo Brevo)
    // Vamos procurar um Header secreto chamado 'auth-token' que configuraremos no painel da Brevo
    const tokenHeader = req.headers.get("auth-token");
    const secret = process.env.BREVO_WEBHOOK_SECRET;

    // Se tivermos um segredo configurado no servidor, validamos se a Brevo nos enviou o mesmo segredo
    if (secret && tokenHeader !== secret) {
      return NextResponse.json({ error: "Acesso não autorizado." }, { status: 401 });
    }

    const payload = await req.json();

    // 2. EXTRAIR DADOS DA BREVO
    const eventType = payload.event; 
    // A Brevo envia as tags como um array de strings. A nossa é a primeira (e única).
    const broadcastId = payload.tags && payload.tags.length > 0 ? payload.tags[0] : null;

    if (!broadcastId) {
      return NextResponse.json({ message: "Evento ignorado: Não pertence a uma campanha da HelloCamp." }, { status: 200 });
    }

    // 3. MAPEAR EVENTOS DA BREVO PARA A BASE DE DADOS
    let updateField = '';
    if (eventType === 'delivered') updateField = 'total_entregues';
    else if (eventType === 'opened' || eventType === 'unique_opened') updateField = 'total_abertos';
    else if (eventType === 'click') updateField = 'total_cliques';
    else if (eventType === 'bounced' || eventType === 'hardBounced' || eventType === 'softBounced' || eventType === 'spam' || eventType === 'invalid') updateField = 'total_falhas';

    // 4. INCREMENTAR AS ESTATÍSTICAS
    if (updateField) {
      const { data: campanha } = await supabaseAdmin
        .from('email_broadcasts')
        .select(updateField)
        .eq('id', broadcastId)
        .single();

      // Sem erros de TypeScript
      const valorAtual = (campanha as Record<string, any>)?.[updateField] || 0;

      await supabaseAdmin
        .from('email_broadcasts')
        .update({ [updateField]: valorAtual + 1 })
        .eq('id', broadcastId);
    }

    return NextResponse.json({ received: true }, { status: 200 });

  } catch (err: any) {
    console.error("Erro no Webhook da Brevo:", err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

import { createClient } from "@supabase/supabase-js";
import { Resend } from "resend";

export const dynamic = "force-dynamic";

  try {
    const body = await req.json();

    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;

        { error: "Falta SUPABASE_SERVICE_ROLE_KEY no ficheiro .env.local." },
        { status: 500 }

    const supabaseAdmin = createClient(supabaseUrl, serviceRoleKey, {
      auth: { autoRefreshToken: false, persistSession: false },

    const cleanEmail = (email || "").trim().toLowerCase();
    if (!cleanEmail) {
      return NextResponse.json(
        { error: "Por favor, introduza um endereço de e-mail válido." },
        { status: 400 }
      );
    }

    // 1. Procurar se o utilizador existe na tabela perfis e em auth.users
    const { data: perfil } = await supabaseAdmin
      .from("perfis")
      .select("id, nome_completo, empresa_nome, role, is_superadmin")
      .ilike("email", cleanEmail)
      .maybeSingle();

    const { data: listUsersData } =
      await supabaseAdmin.auth.admin.listUsers({
        page: 1,
        perPage: 1000,
      });

    const authUser = listUsersData?.users?.find(
      (u) => u.email?.toLowerCase() === cleanEmail
    );

    // =========================================================================
    // AÇÃO 1: ENVIAR MAGIC LINK INSTANTÂNEO VIA RESEND (OU GERAR LINK DE ACESSO)
    // =========================================================================
    if (action === "magic_link") {

        destinoFinal = `${origin}/${lang}/admin/dashboard`;
      } else if (perfil?.role === "monitor") {

          {
            error:
              linkError?.message ||
              "Não foi possível gerar o link mágico para este e-mail.",
          },
          { status: 400 }
        );
      }

      const magicUrl = linkData.properties.action_link;

      // Enviar via Resend para chegar em 1 segundo!

        await resend.emails.send({

import { NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

export const dynamic = "force-dynamic";

export async function GET(req: Request) {
  try {
    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;
    // Usa a Service Role Key para ignorar qualquer bloqueio de RLS na tabela institucional_iniciativas
    const supabaseKey =
      process.env.SUPABASE_SERVICE_ROLE_KEY ||
      process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!;

    const supabaseAdmin = createClient(supabaseUrl, supabaseKey, {
      auth: { autoRefreshToken: false, persistSession: false },
    });

    const { searchParams } = new URL(req.url);
    const id = searchParams.get("id");
    const distrito = searchParams.get("distrito");

    // 1. Se pedir uma iniciativa específica por ID
    if (id) {
      const { data, error } = await supabaseAdmin
        .from("institucional_iniciativas")
        .select("*")
        .eq("id", id)
        .maybeSingle();

      if (error) throw error;
      return NextResponse.json({ data });
    }

    // 2. Lista todas as iniciativas ativas
    let query = supabaseAdmin
      .from("institucional_iniciativas")
      .select("*")
      .order("distrito", { ascending: true })
      .order("concelho", { ascending: true });

    if (distrito) {
      query = query.ilike("distrito", `%${distrito}%`);
    }

    const { data, error } = await query;
    if (error) throw error;

    const ativas = (data || []).filter((item: any) => item.is_active !== false);
    return NextResponse.json({ data: ativas });
  } catch (err: any) {
    console.error("Erro na API /api/institucional:", err);
    return NextResponse.json(
      { error: err.message, data: [] },
      { status: 500 }
    );
  }
}

    return NextResponse.json({ error: "Configurações de servidor em falta." }, { status: 500 });

  const stripe = new Stripe(stripeSecretKey);
  const resend = new Resend(resendApiKey);
  const supabase = createClient(supabaseUrl, supabaseKey);

  try {
    const { reservaId } = await req.json();

    // 1. Busca direta da reserva
    const { data: reserva, error: fetchError } = await supabase
      .from('reservas')
      .select('*')
      .eq('id', reservaId)
      .single();

    if (fetchError || !reserva) {
      throw new Error("Reserva não encontrada na Base de Dados.");
    }

    if (reserva.status_pagamento === 'Reembolsado') {
      throw new Error("Esta reserva já se encontra reembolsada.");
    }

    // 2. Busca plana dos dados complementares
    const { data: campoData } = await supabase.from('campos').select('nome, local').eq('id', reserva.campo_id).single();
    const { data: criancaData } = await supabase.from('criancas').select('nome').eq('id', reserva.crianca_id).single();
    const { data: clienteData } = await supabase.from('perfis').select('nome_completo, email').eq('id', reserva.cliente_id).single();
    const { data: orgData } = await supabase.from('perfis').select('email, empresa_nome, stripe_account_id').eq('id', reserva.organizador_id).single();

    // 3. Localizar a transação correspondente na Stripe
    const sessions = await stripe.checkout.sessions.list({ limit: 100 });
    const matchSession = sessions.data.find(s => {
      if (s.metadata && s.metadata.reservasIds) {
        try {
          const ids = JSON.parse(s.metadata.reservasIds);
          return ids.includes(reservaId);
        } catch (e) { return false; }
      }
      return false;
    });

    if (!matchSession || !matchSession.payment_intent) {

      payment_intent: matchSession.payment_intent as string,

    // Guarda o session_id na reserva para facilitar reembolsos futuros!
    await supabaseAdmin
      .from('reservas')
      .update({ stripe_session_id: session.id })
      .in('id', idsCriados);

    return NextResponse.json({ url: session.url });

  } catch (err: any) {
    console.error("Erro no Checkout:", err);
    return NextResponse.json({ error: err.message }, { status: 500 });
  }
}

function calcularIdade(dataNasc?: string) {
  if (!dataNasc) return 'N/D';
  const hoje = new Date();
  const nasc = new Date(dataNasc);
  if (isNaN(nasc.getTime())) return 'N/D';
  let idade = hoje.getFullYear() - nasc.getFullYear();
  const m = hoje.getMonth() - nasc.getMonth();
  if (m < 0 || (m === 0 && hoje.getDate() < nasc.getDate())) idade--;
  return `${idade} anos`;
}

import { NextResponse } from 'next/server';
import Stripe from 'stripe';
import { Resend } from 'resend';
import { createClient } from '@supabase/supabase-js';

export const dynamic = 'force-dynamic';

export async function POST(req: Request) {
  const stripeSecretKey = process.env.STRIPE_SECRET_KEY;
  const resendApiKey = process.env.RESEND_API_KEY;
  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const supabaseKey = process.env.SUPABASE_SERVICE_ROLE_KEY || process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY;

  if (!stripeSecretKey || !resendApiKey || !supabaseUrl || !supabaseKey) {
    return NextResponse.json({ error: "Configurações de servidor em falta." }, { status: 500 });
  }

  const stripe = new Stripe(stripeSecretKey, { apiVersion: '2023-10-16' as any });
  const resend = new Resend(resendApiKey);
  const supabase = createClient(supabaseUrl, supabaseKey);

  try {
    const { reservaId } = await req.json();

    // 1. Busca da Reserva
    const { data: reserva, error: fetchError } = await supabase
      .from('reservas')
      .select('*')
      .eq('id', reservaId)
      .single();

    if (fetchError || !reserva) throw new Error("Reserva não encontrada na Base de Dados.");
    if (reserva.status_pagamento === 'Reembolsado' || reserva.status_reembolso === 'Reembolsado') {
      throw new Error("Esta reserva já se encontra cancelada e tratada.");
    }

    // 2. Busca de Dados Complementares (incluindo datas do campo para fallback seguro)
    const { data: campoData } = await supabase
      .from('campos')
      .select('nome, local, politica_cancelamento, data_inicio, pacotes, calendario_funcionamento')
      .eq('id', reserva.campo_id)
      .single();
    const { data: criancaData } = await supabase.from('criancas').select('nome').eq('id', reserva.crianca_id).single();
    const { data: clienteData } = await supabase.from('perfis').select('nome_completo, email').eq('id', reserva.cliente_id).single();
    const { data: orgData } = await supabase.from('perfis').select('email, empresa_nome, stripe_account_id').eq('id', reserva.organizador_id).single();

    // 3. MATEMÁTICA: Calcular Dias até ao Início do Campo (Com 5 Níveis de Fallback Seguro)
    let dataInicioTurno: Date | null = null;

    // Fallback 1: Coluna reserva.turno (se existir)
    try {
      const turnoObj = typeof reserva.turno === 'string' ? JSON.parse(reserva.turno) : reserva.turno;
      if (turnoObj?.data_inicio) dataInicioTurno = new Date(turnoObj.data_inicio);
    } catch (e) {}

    // Fallback 2: Primeiro dia escolhido em extras_escolhidos.dias_soltos (formato YYYY-MM-DD)
    if (!dataInicioTurno || isNaN(dataInicioTurno.getTime())) {
      const diasSoltos = reserva.extras_escolhidos?.dias_soltos;
      if (Array.isArray(diasSoltos) && diasSoltos.length > 0) {
        const diasOrdenados = [...diasSoltos].sort();
        const d = new Date(diasOrdenados[0]);
        if (!isNaN(d.getTime())) dataInicioTurno = d;
      }
    }

    // Fallback 3: Calendário de funcionamento do campo ou data_inicio do campo
    if (!dataInicioTurno || isNaN(dataInicioTurno.getTime())) {
      const calInicio = campoData?.calendario_funcionamento?.data_inicio || campoData?.data_inicio;
      if (calInicio) {
        const d = new Date(calInicio);
        if (!isNaN(d.getTime())) dataInicioTurno = d;
      }
    }

    // Fallback 4: Se o campo não tiver data fixa definida (ex: aberto todo o ano), assume 30 dias de antecedência para não bloquear o pai
    if (!dataInicioTurno || isNaN(dataInicioTurno.getTime())) {
      dataInicioTurno = new Date(Date.now() + 30 * 24 * 60 * 60 * 1000);
    }

    const hoje = new Date();
    hoje.setHours(0, 0, 0, 0);
    const inicio = new Date(dataInicioTurno);
    inicio.setHours(0, 0, 0, 0);
    const diffDias = Math.ceil((inicio.getTime() - hoje.getTime()) / (1000 * 60 * 60 * 24));

    // 4. LÓGICA DE CANCELAMENTO (Flexível, Moderada, Estrita)
    const politica = campoData?.politica_cancelamento || 'Flexível';
    const valorPagoGeral = Number(reserva.valor_pago) || Number(reserva.valor_total) || 0;
    let valorAReembolsar = 0;
    let descricaoPolitica = 'Estrita (Sem Reembolso)';

    if (politica.includes('Flexível')) {
      if (diffDias >= 7) {
        valorAReembolsar = valorPagoGeral;
        descricaoPolitica = 'Flexível (100% devolvido)';
      } else {
        descricaoPolitica = `Flexível (Cancelado a menos de 7 dias: ${diffDias} dias)`;
      }
    } else if (politica.includes('Moderada')) {
      if (diffDias >= 15) {
        valorAReembolsar = valorPagoGeral * 0.5;
        descricaoPolitica = 'Moderada (50% devolvido)';
      } else {
        descricaoPolitica = `Moderada (Cancelado a menos de 15 dias: ${diffDias} dias)`;
      }
    } else if (politica.includes('Estrita')) {
      valorAReembolsar = 0;
      descricaoPolitica = 'Estrita (0% devolvido)';
    }

    // 5. EXECUTAR REEMBOLSO NA STRIPE (Apenas se houver valor a devolver)
    let paymentIntentId = reserva.stripe_payment_intent_id;

    if (valorAReembolsar > 0) {
      if (!paymentIntentId && reserva.stripe_session_id) {
        const session = await stripe.checkout.sessions.retrieve(reserva.stripe_session_id);
        paymentIntentId = session.payment_intent as string;
      }
      if (!paymentIntentId) {
        const sessions = await stripe.checkout.sessions.list({ limit: 100 });
        const matchSession = sessions.data.find(s => {
          if (s.metadata && s.metadata.reservasIds) {
            try { return JSON.parse(s.metadata.reservasIds).includes(reservaId); } catch (e) { return false; }
          }
          return false;
        });
        paymentIntentId = matchSession?.payment_intent as string;
      }

      if (!paymentIntentId) {
        throw new Error("Transação Stripe não encontrada. O reembolso financeiro terá de ser manual.");
      }

      const quantiaReembolsoCentimos = Math.round(valorAReembolsar * 100);
      const refundParams: Stripe.RefundCreateParams = {
        payment_intent: paymentIntentId,
        amount: quantiaReembolsoCentimos,
        reason: 'requested_by_customer'
      };

      if (orgData?.stripe_account_id) {
        refundParams.reverse_transfer = true;
        if (valorAReembolsar === valorPagoGeral) {
          refundParams.refund_application_fee = true;
        }
      }

      await stripe.refunds.create(refundParams);
    }

    // 6. ATUALIZAR A BASE DE DADOS
    await supabase.from('reservas').update({
      status_pagamento: 'Reembolsado',
      status_reembolso: 'Reembolsado',
      status: 'Reembolsado',
      dados_reembolso: {
        data_estorno: new Date().toISOString(),
        processado_por: 'Automático via Cliente',
        politica_aplicada: descricaoPolitica,
        valor_devolvido: valorAReembolsar
      }
    }).eq('id', reservaId);

    // 7. ENVIO DOS EMAILS
    const campoNome = campoData?.nome || 'Programa HelloCamp';
    const nomeCrianca = criancaData?.nome || 'Participante';
    const turnoNome = reserva.turno_nome || 'Programa Base';

    const nomePai = clienteData?.nome_completo || reserva.nome_encarregado || 'Encarregado de Educação';

    if (emailPai) {
      await resend.emails.send({
        from: 'HelloCamp Finanças <info@hellocamp.pt>',
        to: emailPai,
        subject: `Confirmação de Cancelamento - ${campoNome}`,
        html: `

            <div style="background-color: #fef2f2; border: 1px solid #fca5a5; border-radius: 12px; padding: 25px; text-align: center;">
              <h2 style="font-size: 18px; font-weight: 700; margin-top: 0; color: #991b1b;">Inscrição Cancelada com Sucesso</h2>

const SENDER_NAME = 'Rodrigo | HelloCamp';

async function enviarViaBrevo(params: {
  to: { email: string; name?: string };
  subject: string;
  html: string;
}): Promise<{ success: boolean; messageId?: string; error?: any }> {
  if (!BREVO_API_KEY) return { success: false, error: { message: 'Sem API Key' } };

  try {
    const res = await fetch(BREVO_ENDPOINT, {
      method: 'POST',
      headers: {
        'accept': 'application/json',
        'api-key': BREVO_API_KEY,
        'content-type': 'application/json',
      },
      body: JSON.stringify({
        sender: { name: SENDER_NAME, email: SENDER_EMAIL },
        to: [params.to],
        subject: params.subject,
        htmlContent: params.html,
        headers: { 'X-Mailer': 'HelloCamp-B2B-Global' },
      }),
    });

    const data = await res.json().catch(() => ({}));
    if (!res.ok) return { success: false, error: data };
    return { success: true, messageId: data?.messageId };
  } catch (err: any) {
    return { success: false, error: { message: String(err) } };
  }
}

export async function GET(request: Request) {
  try {
    const authHeader = request.headers.get('authorization');
    if (process.env.CRON_SECRET && authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
      return NextResponse.json({ error: 'Não autorizado' }, { status: 401 });
    }

    const seteDiasAtras = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000).toISOString();

    // SEGURANÇA 1: Vamos buscar os e-mails de todos os Parceiros já registados
    const { data: perfisAtivos } = await supabaseAdmin
      .from('perfis')
      .select('email')
      .eq('role', 'organizador');

    const emailsRegistados = perfisAtivos?.map(p => p.email?.toLowerCase().trim()) || [];

    // Busca as leads na fila
    const { data: leadsPotenciais, error } = await supabaseAdmin
      .from('leads_externas')
      .select('id, nome_cliente, email_cliente, fase_funil, ultimo_contacto, idioma')
      .is('unsubscribed', false)
      .lt('fase_funil', 4)
      .or(`ultimo_contacto.is.null,ultimo_contacto.lt.${seteDiasAtras}`)
      .not('email_cliente', 'is', null)
      .neq('email_cliente', '')
      .order('fase_funil', { ascending: true })
      .order('created_at', { ascending: true })
      .limit(300);

    if (error) throw error;
    if (!leadsPotenciais || leadsPotenciais.length === 0) {
      return NextResponse.json({ message: 'Nenhuma lead elegível hoje.' });
    }

    // SEGURANÇA 2: Filtra as leads para garantir que nenhuma está na lista de parceiros registados
    const leadsProntas: any[] = [];
    const leadsParaDesativar: string[] = [];

    for (const lead of leadsPotenciais) {
      const email = lead.email_cliente?.toLowerCase().trim();
      if (emailsRegistados.includes(email)) {
        leadsParaDesativar.push(lead.id);
      } else {
        leadsProntas.push(lead);
      }
    }

    if (leadsParaDesativar.length > 0) {
      await supabaseAdmin.from('leads_externas')
        .update({ status: 'COMPLETO', unsubscribed: true, fase_funil: 4 })
        .in('id', leadsParaDesativar);
    }

    if (leadsProntas.length === 0) {
      return NextResponse.json({ message: 'Leads filtradas porque já eram parceiros registados. Nenhuma enviada.' });
    }

    const leadsFinal = leadsProntas.slice(0, 250);
    const resultados: any[] = [];
    const BATCH_SIZE = 25;

    for (let i = 0; i < leadsFinal.length; i += BATCH_SIZE) {
      const batch = leadsFinal.slice(i, i + BATCH_SIZE);

      const batchResults = await Promise.all(
        batch.map(async (lead) => {
          const email = (lead.email_cliente || '').trim();
          const nome = lead.nome_cliente || 'Organizador';
          const faseAtual = Number(lead.fase_funil) || 0;

          let lang = (lead.idioma || 'PT').toUpperCase();
          if (!TEMPLATES[lang]) lang = 'EN';

          const tpl = TEMPLATES[lang][faseAtual];
          if (!tpl) return { status: 'skipped', id: lead.id };

          // CORREÇÃO CRÍTICA: O site só suporta rotas /pt e /en, e a página real é /parceiro (singular)
          const linkLang = lang === 'PT' ? 'pt' : 'en';
          const ctaLinkTracker = `${BASE_URL}/${linkLang}/parceiro?utm_source=cold_email_b2b&utm_medium=email&utm_campaign=fase_${faseAtual}_${lang}`;
          const unsubscribeUrl = `${BASE_URL}/${linkLang}/unsubscribe?id=${lead.id}`;

          const r = await enviarViaBrevo({
            to: { email, name: nome },
            subject: tpl.gerarAssunto(nome),
            html: getTemplateHTML(tpl.titulo, tpl.gerarMensagem(nome), tpl.textoCta, ctaLinkTracker, tpl.gerarFooter(unsubscribeUrl)),
          });

          if (r.success) {
            return { status: 'sent', id: lead.id, novaFase: faseAtual + 1 };
          }
          return { status: 'failed', id: lead.id };
        })
      );

      resultados.push(...batchResults);
    }

    const sucessos = resultados.filter(r => r.status === 'sent');

    if (sucessos.length > 0) {
      const groupedByFase = sucessos.reduce((acc: any, curr: any) => {
        if (!acc[curr.novaFase]) acc[curr.novaFase] = [];
        acc[curr.novaFase].push(curr.id);
        return acc;
      }, {} as Record<number, string[]>);

      const nowIso = new Date().toISOString();
      await Promise.all(
        Object.entries(groupedByFase).map(([faseStr, ids]) =>
          supabaseAdmin.from('leads_externas')
            .update({ fase_funil: parseInt(faseStr, 10), ultimo_contacto: nowIso })
            .in('id', ids as string[])
        )
      );
    }

    return NextResponse.json({
      success: true,
      enviados: sucessos.length,
      tentados: leadsFinal.length,
      limposPorJaSeremParceiros: leadsParaDesativar.length
    });

  } catch (error: any) {
    console.error('❌ Erro no Cron B2B Global:', error);
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

      });

      const resendData = await resendRes.json();
      if (!resendRes.ok) {
        return NextResponse.json(
          { error: resendData?.message || "Erro ao enviar via Resend." },
          { status: 400 }
        );
      }

      return NextResponse.json({ success: true, id: resendData.id });
    }

    return NextResponse.json({ error: "Ação inválida." }, { status: 400 });
  } catch (err: any) {
    return NextResponse.json(
      { error: err.message || "Erro interno do servidor." },
      { status: 500 }
    );
  }

import { NextResponse } from "next/server";
import { createClient } from "@supabase/supabase-js";

export const dynamic = "force-dynamic";

export async function POST(req: Request) {
  try {
    const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!;
    const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY!;

    if (!supabaseUrl || !serviceRoleKey) {
      return NextResponse.json(
        { error: "Configuração SUPABASE_SERVICE_ROLE_KEY em falta no servidor." },
        { status: 500 }
      );
    }

    const supabaseAdmin = createClient(supabaseUrl, serviceRoleKey, {
      auth: { autoRefreshToken: false, persistSession: false },
    });

    const formData = await req.formData();
    const id = (formData.get("id") as string | null)?.trim() || "";
    const emailFromForm =
      ((formData.get("emailAcesso") || formData.get("email")) as string | null)?.trim() || "";
    const passwordNova = (formData.get("passwordNova") as string | null) || "";
    const contratoDadosStr = (formData.get("contratoDados") as string | null) || "";
    const lang = ((formData.get("lang") as string | null) || "pt").toLowerCase();
    const pdfFile = formData.get("pdf") as File | null;

    // 1. Verificar exatamente que parâmetro possa estar em falta
    const faltam: string[] = [];
    if (!id) faltam.push("ID do Parceiro");
    if (!contratoDadosStr) faltam.push("Dados do Contrato");
    if (!pdfFile) faltam.push("Ficheiro PDF do Contrato");

    if (faltam.length > 0) {
      return NextResponse.json(
        {
          error: `Dados incompletos submetidos no formulário. Em falta: ${faltam.join(", ")}.`,
        },
        { status: 400 }
      );
    }

    const contratoDados = JSON.parse(contratoDadosStr);

    // 2. Carregar o perfil na base de dados
    const { data: perfil, error: fetchError } = await supabaseAdmin
      .from("perfis")
      .select("email, email_publico, contrato_dados, taxa_comissao, base_comissao")
      .eq("id", id)
      .maybeSingle();

    if (fetchError || !perfil) {
      return NextResponse.json(
        { error: "Perfil de parceiro não encontrado na base de dados." },
        { status: 404 }
      );
    }

    // Determinar o e-mail final com fallback seguro
    const emailFinal = (
      contratoDados.emailContacto ||
      emailFromForm ||
      perfil.email ||
      perfil.email_publico ||
      ""
    )
      .trim()
      .toLowerCase();

    if (!emailFinal) {
      return NextResponse.json(
        { error: "Falta indicar o E-mail Comercial / Login no formulário." },
        { status: 400 }
      );
    }

    // 3. Se foi definida uma nova password (fluxo de Onboarding), atualiza ou cria o utilizador em auth.users
    let loginReady = false;
    if (passwordNova && passwordNova.length >= 6) {
      const { error: updateErr } =
        await supabaseAdmin.auth.admin.updateUserById(id, {
          email: emailFinal,
          password: passwordNova,
          email_confirm: true,
          user_metadata: {
            role: "organizador",
            empresa_nome:
              contratoDados.empresaNome || contratoDados.nomeEmpresa || "",
          },
        });

      if (!updateErr) {
        loginReady = true;
      } else {
        const { error: createErr } =
          await supabaseAdmin.auth.admin.createUser({
            id,
            email: emailFinal,
            password: passwordNova,
            email_confirm: true,
            user_metadata: {
              role: "organizador",
              empresa_nome:
                contratoDados.empresaNome || contratoDados.nomeEmpresa || "",
            },
          });

        if (!createErr) {
          loginReady = true;
        } else {
          // Fallback via RPC se existir
          const { error: rpcError } = await supabaseAdmin.rpc(
            "admin_set_partner_password",
            {
              target_user_id: id,
              new_password: passwordNova,
              target_email: emailFinal,
            }
          );
          if (!rpcError) loginReady = true;
        }
      }
    }

    // 4. Upload do Documento PDF assinado para o Storage
    const fileName = `contrato_global_${id}_${Date.now()}.pdf`;
    const arrayBuffer = await pdfFile!.arrayBuffer();
    const buffer = Buffer.from(arrayBuffer);

    let contratoUrl = `https://www.hellocamp.pt/${lang}/assinatura-contrato/${id}`;

    const { error: uploadError } = await supabaseAdmin.storage
      .from("campos-documentos")
      .upload(fileName, buffer, {
        contentType: "application/pdf",
        upsert: true,
      });

    if (!uploadError) {
      const { data: publicUrlData } = supabaseAdmin.storage
        .from("campos-documentos")
        .getPublicUrl(fileName);
      if (publicUrlData?.publicUrl) {
        contratoUrl = publicUrlData.publicUrl;
      }
    } else {
      console.warn(
        "Aviso Storage campos-documentos:",
        uploadError.message
      );
    }

    const nomeEmpresaFinal =
      contratoDados.empresaNome ||
      contratoDados.nomeEmpresa ||
      "Parceiro HelloCamp";
    const taxaFinal =
      contratoDados.taxaComissao ?? perfil.taxa_comissao ?? 12;
    const baseFinal =
      contratoDados.baseComissao || perfil.base_comissao || "total";

    // 5. Atualizar a Tabela 'perfis' com Aprovação e Dados Fiscais Completos
    const { error: updatePerfilError } = await supabaseAdmin
      .from("perfis")
      .update({
        email: emailFinal,
        email_publico: contratoDados.emailReservas || emailFinal,
        role: "organizador",
        empresa_nome: nomeEmpresaFinal,
        nome_empresa: nomeEmpresaFinal,
        nome: nomeEmpresaFinal,
        nif_empresa: contratoDados.nif || null,
        nome_completo:
          contratoDados.pessoaContacto ||
          contratoDados.assinaturaNome ||
          nomeEmpresaFinal,
        telefone: contratoDados.telefone || null,
        telefone_publico: contratoDados.telefone || null,
        morada_sede: contratoDados.morada || null,
        website: contratoDados.website || null,
        status_contrato: "Aprovado",
        parceiro_verificado: true,
        verificado: true,
        contrato_dados: contratoDados,
        taxa_comissao: taxaFinal,
        base_comissao: baseFinal,
        modalidade_reserva: contratoDados.modalidadeReserva || "direta",
        link_externo_reserva:
          contratoDados.modalidadeReserva === "link_externo"
            ? contratoDados.linkExternoReserva
            : null,
      })
      .eq("id", id);

    if (updatePerfilError) {
      throw new Error(
        `Erro ao atualizar perfil: ${updatePerfilError.message}`
      );
    }

    // 6. Ativar todos os campos associados ao organizador
    await supabaseAdmin
      .from("campos")
      .update({
        status_aprovacao: "Aprovado",
        ativo: true,
        contrato_dados: contratoDados,
        contrato_parceiro_url: contratoUrl,
        taxa_comissao: taxaFinal,
        base_comissao: baseFinal,
        modalidade_reserva: contratoDados.modalidadeReserva || "direta",
        link_externo_reserva:
          contratoDados.modalidadeReserva === "link_externo"
            ? contratoDados.linkExternoReserva
            : null,
        tipo_pagamento:
          contratoDados.modalidadeReserva !== "link_externo"
            ? contratoDados.tipoPagamento ||
              contratoDados.tipo_pagamento ||
              "100_total"
            : null,
        politica_cancelamento:
          contratoDados.modalidadeReserva !== "link_externo"
            ? contratoDados.politicaCancelamento ||
              contratoDados.politica_cancelamento ||
              "Moderada (Reembolso a 50% até 15 dias antes)"
            : null,
      })
      .eq("organizador_id", id);

    return NextResponse.json({
      success: true,
      url: contratoUrl,
      loginReady,
    });
  } catch (error: any) {
    console.error("Erro na API Assinar Contrato:", error.message);
    return NextResponse.json(
      { error: error.message || "Erro interno ao assinar contrato." },
      { status: 500 }
    );
  }
}
